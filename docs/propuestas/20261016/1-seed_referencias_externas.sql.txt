-- ---------------------------------------------------------------------
-- Tilapia: etapas tempranas (pesos inferidos del encadenamiento con Desarrollo=20 g y FAO)
-- Los % de la foto borrosa siguen por confirmar.
-- ---------------------------------------------------------------------
update etapas_alimentacion e
set peso_min_g = v.pmin, peso_max_g = v.pmax,
    referencia_externa_id = (select id from fuentes where codigo = 'ext-fao-tabla-28-tilapia'),
    notas = 'Pesos propuestos con la referencia FAO (0-1, 1-5, 5-20 g); no leídos en la foto. FAO: 30-10 %, 10-6 % y 6-4 % en intensivo.'
from (values
  ('Etapa temprana 1 (por confirmar)', 0, 1),
  ('Etapa temprana 2 (por confirmar)', 1, 5),
  ('Etapa temprana 3 (por confirmar)', 5, 20)
) as v(etapa, pmin, pmax)
where e.etapa = v.etapa
  and e.especie_id = (select id from especies where codigo = 'tilapia');

-- ---------------------------------------------------------------------
-- Amoniaco: se reemplaza la tabla pendiente por la ecuación de Emerson et al. (1975)
-- pKa = 0.09018 + 2729.92 / T(K);  fracción NH3 = 1 / (10^(pKa - pH) + 1)
-- Dos documentos imprimen 2727.92 y 2729.92; la diferencia es menor al 1 % en el resultado.
-- Casos de prueba: TAN 1, pH 8.5, 29 °C -> ~0.19 ; TAN 1, pH 7.0, 25 °C -> ~0.0057
-- ---------------------------------------------------------------------
update formulas
set expresion = 'tan / (10 ^ (0.09018 + 2729.92 / (273.15 + temp_c) - ph) + 1)',
    variables = '[{"nombre":"tan","unidad":"mg N/L"},{"nombre":"temp_c","unidad":"°C","descripcion":"válido de 0 a 50 °C"},{"nombre":"ph"}]'::jsonb,
    unidad_resultado = 'mg NH3-N/L',
    validacion = 'confirmado',
    referencia_externa_id = (select id from fuentes where codigo = 'ext-emerson-1975-amoniaco'),
    notas = 'Ecuación de Emerson et al. (1975), base de la tabla UF/IFAS. Casos de prueba: (1, 8.5, 29) ~ 0.19; (1, 7.0, 25) ~ 0.0057. Después convertir a NH3 con acui_nh3 (x 17/14).'
where codigo = 'acui_nh3n';

-- ---------------------------------------------------------------------
-- Tilapia: reglas informativas de referencia externa (solo alertan, no cambian la ración)
-- ---------------------------------------------------------------------
insert into reglas_alerta (especie_id, parametro_id, operador, umbral, severidad, factor_racion, mensaje, fuente_id, validacion, referencia_externa_id)
select e.id, p.id, v.op, v.umbral, 'alerta'::estado_rango, null, v.msg, f.id, 'por_confirmar'::validacion_estado, f.id
from (values
  ('<', 17, 'Referencia externa: alrededor de 16-17 °C la tilapia deja de comer de forma natural', 'ext-nepjol-tilapia'),
  ('>', 32, 'Referencia externa: sobre 32 °C baja la capacidad de digestión; vigilar O2 y valorar suspender la toma de la tarde', 'ext-frontiers-tilapia-temperatura')
) as v(op, umbral, msg, fuente)
join especies e on e.codigo = 'tilapia'
join parametros p on p.codigo = 'temperatura_agua'
join fuentes f on f.codigo = v.fuente;

-- Nota: tilapia tolera O2 muy bajo por periodos cortos (referencia externa). Los umbrales de 3.0 y 2.0 mg/L
-- de clase son umbrales de manejo conservador, no límites letales. Se mantienen tal cual.
update reglas_alerta
set referencia_externa_id = (select id from fuentes where codigo = 'ext-springer-tilapia')
where parametro_id = (select id from parametros where codigo = 'oxigeno_disuelto')
  and especie_id = (select id from especies where codigo = 'tilapia');

set validacion = 'confirmado',
    referencia_externa_id = (select id from fuentes where codigo = 'ext-lafeber-ajolote')

update rangos_parametro

where especie_id = (select id from especies where codigo = 'ajolote')
  and parametro_id = (select id from parametros where codigo = 'nitrato');

insert into rangos_parametro (especie_id, parametro_id, estado, valor_min, valor_max, max_inclusivo, accion, fuente_id, validacion, referencia_externa_id)
select e.id, p.id, v.estado::estado_rango, v.vmin, v.vmax, true, v.accion, f.id, 'por_confirmar'::validacion_estado, f.id
from (values
  ('ph', 'normal', 6.5, 7.4, 'Aceptable; evitar cambios bruscos de pH', 'ext-azeah-ajolote'),
  ('ph', 'normal', 7.6, 8.0, 'Aceptable; evitar cambios bruscos de pH', 'ext-azeah-ajolote'),
  ('temperatura_agua', 'alerta', 20, null, 'Estrés y mayor riesgo de enfermedad con agua tibia', 'ext-guias-cuidado-ajolote')
) as v(param, estado, vmin, vmax, accion, fuente)
join especies e on e.codigo = 'ajolote'
join parametros p on p.codigo = v.param
join fuentes f on f.codigo = v.fuente;
